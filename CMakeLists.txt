cmake_minimum_required(VERSION 3.22)
project(BrezeBass VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BREZEBASS_CORE_ONLY "DSP tests without JUCE" OFF)
enable_testing()
add_executable(brezebass_core_tests CoreTests.cpp)
target_include_directories(brezebass_core_tests PRIVATE .)
add_test(NAME CoreDSP COMMAND brezebass_core_tests)
if(BREZEBASS_CORE_ONLY)
 return()
endif()
include(FetchContent)
FetchContent_Declare(JUCE URL https://github.com/juce-framework/JUCE/archive/refs/tags/8.0.6.tar.gz
 URL_HASH SHA256=95752f8e8b37d802b4c10182bc757de7d88f0c8899e879afa8798589be306a11)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(BrezeBass COMPANY_NAME "BrezeBass Audio" BUNDLE_ID "audio.brezebass.synth"
 PLUGIN_MANUFACTURER_CODE Brze PLUGIN_CODE Brz3 PRODUCT_NAME "BREZEBASS"
 FORMATS VST3 Standalone IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
 VST3_CATEGORIES Instrument Synth COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(BrezeBass PRIVATE Processor.cpp Editor.cpp)
target_include_directories(BrezeBass PUBLIC .)
target_compile_definitions(BrezeBass PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(BrezeBass PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
juce_add_console_app(brezebass_state_tests PRODUCT_NAME "brezebass_state_tests")
target_sources(brezebass_state_tests PRIVATE StateTests.cpp Processor.cpp Editor.cpp)
target_include_directories(brezebass_state_tests PRIVATE .)
target_compile_definitions(brezebass_state_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(brezebass_state_tests PRIVATE juce::juce_audio_utils)
add_test(NAME PluginState COMMAND brezebass_state_tests)
juce_add_console_app(brezebass_host_smoke PRODUCT_NAME "brezebass_host_smoke")
target_sources(brezebass_host_smoke PRIVATE HostSmoke.cpp)
target_compile_definitions(brezebass_host_smoke PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(brezebass_host_smoke PRIVATE juce::juce_audio_utils)
